const CACHE_NAME = 'lensaamal-v1'; const urlsToCache = [ './', './index.html', './icon.png', './manifest.json' ]; self.addEventListener('install', event => { event.waitUntil( caches.open(CACHE_NAME) .then(cache => { return cache.addAll(urlsToCache); }) ); }); self.addEventListener('fetch', event => { event.respondWith( caches.match(event.request) .then(response => { if (response) { return response; // Gunakan versi cache jika offline } return fetch(event.request); // Ambil dari internet jika online }) ); });
